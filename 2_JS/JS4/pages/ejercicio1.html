// renderizo los usuarios como tarjetas function renderUsuarios(usuarios) { const contenedor = document.getElementById('output-box'); contenedor.innerHTML = ''; // Le quitamos el padding para que la grilla ocupe todo el ancho sin verse encerrada en la caja si hay resultados contenedor.style.padding = '0'; contenedor.style.border = 'none'; contenedor.style.background = 'transparent'; if (!usuarios || usuarios.length === 0) { contenedor.style.padding = '2rem'; contenedor.style.border = '1px dashed var(--border)'; contenedor.style.background = 'var(--bg-secondary)'; contenedor.innerHTML = '
No hay usuarios para mostrar
'; return; } const grid = document.createElement('div'); grid.className = 'users-grid'; usuarios.forEach(usuario => { const card = document.createElement('div'); card.className = 'user-card'; card.innerHTML = `
${getIniciales(usuario.nombre)}
${usuario.nombre}
${usuario['correo electrónico']}
`; grid.appendChild(card); }); contenedor.appendChild(grid); } // muestro un estado de carga function mostrarCargando() { const contenedor = document.getElementById('output-box'); contenedor.style.padding = '2rem'; contenedor.style.border = '1px dashed var(--border)'; contenedor.style.background = 'var(--bg-secondary)'; contenedor.innerHTML = `
Cargando usuarios...
`; } // click en el botón de carga único document.getElementById('btn-cargar').addEventListener('click', async () => { const btn = document.getElementById('btn-cargar'); const metodo = document.querySelector('input[name="metodo_carga"]:checked').value; btn.disabled = true; btn.innerHTML = ' Cargando...'; mostrarCargando(); limpiarConsole('consola-ej1'); logConsole('consola-ej1', `Iniciando request con ${metodo.toUpperCase()}...`, 'info'); logConsole('consola-ej1', `Endpoint: GET ${API_URL}`, 'info'); try { let usuarios = []; if (metodo === 'fetch') { const respuesta = await fetch(API_URL); logConsole('consola-ej1', `Response status: ${respuesta.status} ${respuesta.statusText}`, 'success'); if (!respuesta.ok) throw new Error(`Error HTTP: ${respuesta.status}`); logConsole('consola-ej1', 'Parseando JSON (respuesta.json())...', 'info'); usuarios = await respuesta.json(); } else { const respuesta = await axios.get(API_URL); logConsole('consola-ej1', `Response status: ${respuesta.status} ${respuesta.statusText}`, 'success'); logConsole('consola-ej1', 'Axios parsea el JSON automaticamente (.data)', 'info'); usuarios = respuesta.data; } logConsole('consola-ej1', `Se recibieron ${usuarios.length} usuarios. Renderizando UI.`, 'success'); renderUsuarios(usuarios); mostrarToast(`Datos cargados usando ${metodo}`, 'success'); } catch (error) { logConsole('consola-ej1', `Excepcion atrapada: ${error.message}`, 'error'); const contenedor = document.getElementById('output-box'); contenedor.style.padding = '2rem'; contenedor.style.border = '1px dashed var(--error)'; contenedor.style.background = 'var(--error-bg)'; contenedor.innerHTML = `
Error al obtener los datos: ${error.message}
`; mostrarToast('Ocurrió un error', 'error'); } finally { btn.disabled = false; btn.innerHTML = ' Cargar usuarios'; logConsole('consola-ej1', 'Proceso terminado.', 'info'); } }); // resetear al cambiar el metodo document.querySelectorAll('input[name="metodo_carga"]').forEach(radio => { radio.addEventListener('change', () => { const contenedor = document.getElementById('output-box'); contenedor.style.padding = '2rem'; contenedor.style.border = '1px dashed var(--border)'; contenedor.style.background = 'var(--bg-secondary)'; contenedor.innerHTML = '
Hacé clic en el botón para obtener los usuarios
'; limpiarConsole('consola-ej1'); logConsole('consola-ej1', `Metodo cambiado a ${radio.value.toUpperCase()}. Listo para cargar.`, 'info'); }); });
